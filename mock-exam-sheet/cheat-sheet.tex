\section*{TMA4268 Cheat Sheet}

\section{Bias--variance \& double descent}
$\E[(y_0-\hat f(x_0))^2] = \underbrace{(f(x_0)-\E[\hat f(x_0)])^2}_{\text{Bias}^2} + \Var(\hat f(x_0)) + \sigma^2$.
Outer $\E$: over training set \emph{and} test noise $\varepsilon_0$. Inner Bias/Var: training set only.
Cross-terms vanish: (i) $\varepsilon_0\!\perp\!\hat f$, $\E\varepsilon=0$; (ii) $\E[\hat f-\E\hat f]=0$.
Identity is \emph{exact} for any estimator. \tr{double descent}: in $p\!\gg\! n$, SGD/pseudoinverse picks min-$\|\theta\|_2$ interpolator $\Rightarrow$ implicit ridge $\Rightarrow$ variance \emph{can fall} past interpolation peak. Bias+Var sum still holds.
Prof's rename: ``avoid bad overfitting'' (not trade-off): regularization cuts var without paying full bias.

\hd{Shrinkage MSE} For $\hat\mu_c=c\bar y$: $c^\star = \mu^2/(\mu^2+\sigma^2/n)$.

\section{Linear regression}
$\hat\beta=(X^\top X)^{-1}X^\top y$. $\Var(\hat\beta)=\sigma^2(X^\top X)^{-1}$.
$\df_{\text{model}}=\#\text{params incl.\ intercept}$; $K$-level factor $\to K-1$ dummies.
\hd{CI vs PI} CI: $\hat y_0\pm t\hat\sigma\sqrt{x_0^\top(X^\top X)^{-1}x_0}$; PI: $\hat y_0\pm t\hat\sigma\sqrt{1+x_0^\top(X^\top X)^{-1}x_0}$ (\,$+1$ = irreducible). $\approx 1.96$ for large $n-p-1$.
\hd{$R^2$ vs adj} $R^2_{\text{adj}}=1-(1-R^2)\tfrac{n-1}{n-p-1}$. $R^2$ monotone in $p$; adj may drop; large gap $\Rightarrow$ over-parameterized.
\hd{MLE $\equiv$ LS} under $\varepsilon\sim N(0,\sigma^2)$:
$\log L = -\tfrac{n}{2}\log(2\pi\sigma^2) - \tfrac{1}{2\sigma^2}\sum(y_i-x_i^\top\beta)^2$, drop constants $\Rightarrow \arg\max=\arg\min$ RSS. Laplace errors $\Rightarrow$ L1 (LAD), robust.
\hd{Diagnostics} resid vs fit (linearity/var); QQ (normality \tr{prof favors}); scale-loc (heterosc.); resid vs leverage. Fanning $\Rightarrow$ log-$y$ or WLS.

\section{Collinearity}
\textbf{Perfect}: $X^\top X$ singular $\Rightarrow$ OLS \emph{undefined} (\tr{not ``infinite SE''}); software drops a column.
\textbf{Near}: $(X^\top X)^{-1}$ inflates diagonal in collinear direction; SE explode but $\hat y=X\hat\beta$ stable. Pairs have large negative $\Cov(\hat\beta_j,\hat\beta_k)$ that cancels in joint contribution.
\hd{Fingerprint} individually insig.\ $t$ + significant joint $F$ + drop-one $\approx$ same test MSE.
\tr{Standardization does NOT fix collinearity} (rank unchanged); only helps shrinkage scale-fairness.
Ridge fix: $(X^\top X+\lambda I)^{-1}$ exists $\forall\lambda>0$; splits coef across collinear pair. Lasso: corner $\Rightarrow$ picks one. VIF $\approx 1/(1-R_j^2)$.

\section{Interactions \& categorical}
With $x{:}z$ in model: $\beta_z=$ effect of $z$ \emph{at $x=0$} only (\tr{often outside data}; centre $x$ for sane interpretation).
Slope of $x$ at level $z$: $\beta_x+\beta_{xz}z$. \textbf{Hierarchy principle}: keep main effects whenever interaction stays in.
Re-encoding/reference flip: RSS, fits, $\hat p$ \emph{invariant}; coefficients and SEs change.
\tr{Dummy trap}: $K$ dummies + intercept $\Rightarrow$ singular (sum to 1). Use $K-1$ dummies. Drop one; ridge does \emph{not} fix this structurally.

\section{Logistic regression}
$\eta=\beta^\top x$, $\hat p=\sigma(\eta)=\tfrac{e^\eta}{1+e^\eta}$. Odds $=p/(1-p)$; $p=O/(1+O)$.
$\exp(\beta_j\Delta)=$ odds-ratio per $\Delta x_j$. \tr{$\beta_j$ is on log-odds, NOT probability}: $\partial p/\partial x_j=\beta_j p(1-p)$; max slope $\approx 0.25\beta$ at $p=0.5$.
$p_{\text{new}}=\tfrac{OR\cdot p}{1-(1-OR)p}$ (only $\approx OR\cdot p$ for tiny $p$).
\hd{Interaction $x{:}z$} odds factor per $\Delta x{=}1$ at level $z$: $\exp(\beta_x+\beta_{xz}z)$ (per-group multipliers; sign can flip with $z$). Predicted $p$ under group switch: add \emph{both} dummy and interaction terms.

\section{LDA / QDA / Bayes}
$\delta_k(x)=x^\top\Sigma^{-1}\mu_k -\tfrac12\mu_k^\top\Sigma^{-1}\mu_k+\log\pi_k$ (LDA: shared $\Sigma$, drops $-\tfrac12 x^\top\Sigma^{-1}x$ as $k$-free $\Rightarrow$ linear boundary).
QDA: $\delta_k=-\tfrac12\log|\Sigma_k|-\tfrac12(x-\mu_k)^\top\Sigma_k^{-1}(x-\mu_k)+\log\pi_k$ (per-class $\Sigma_k$ $\Rightarrow$ quadratic boundary).
\hd{2D boundary recipe} $f_A\pi_A=f_B\pi_B$: $v=\Sigma^{-1}(\mu_A-\mu_B);\ c=-\tfrac12\mu_A^\top\Sigma^{-1}\mu_A+\tfrac12\mu_B^\top\Sigma^{-1}\mu_B+\log(\pi_A/\pi_B)$; line $v^\top x+c=0$.
Sanity: equal priors $\Rightarrow$ midpoint on line.
\hd{Prior shift} $\pi_k\uparrow$ moves boundary \emph{away} from class $k$ (its decision region \emph{grows}); equivalently, more area labelled $k$. Boundary shift along $v$: $\tfrac{1}{\|v\|^2}\log(\pi_A/\pi_B)$.
\textbf{Small $n$}: LDA $>$ QDA (per-class $\Sigma_k$ has $Kp(p+1)/2$ params, noisy). Binary features violate Gaussian assumption $\Rightarrow$ logistic safer.
\textbf{Naive Bayes} $\equiv$ diagonal-$\Sigma$ LDA.
\tr{Mixture class density}: $f_0=\sum w_m\phi(\cdot;\mu_m,\Sigma)$ $\Rightarrow$ boundary neither linear nor quadratic (log of sum-of-exps doesn't collapse) -- LDA/QDA fail.

\section{Resampling: CV, bootstrap}
\textbf{$k$-fold} $\df_k=\tfrac{1}{k}\sum_k\!\text{MSE}_k$, $\widehat{\text{SE}}=\text{sd}(\text{MSE}_k)/\sqrt k$ (\tr{``not quite valid''}: folds share data).
\textbf{LOOCV}: \tr{low bias, HIGH variance} (overlapping train sets correlate fold errors). $k{=}5,10$ usually wins on variance.
\hd{OLS LOOCV shortcut} $\text{CV}_n=\tfrac1n\sum\!\bigl(\tfrac{y_i-\hat y_i}{1-h_{ii}}\bigr)^2$, $H=X(X^\top X)^{-1}X^\top$. OLS-only.
\hd{1-SE rule} $\lambda_{1\text{SE}}=\max\{\lambda:\text{CV}(\lambda)\le\text{CV}(\hat\lambda_{\min})+\widehat{\text{SE}}\}$; pick \emph{simplest} (\tr{$\lambda\uparrow$, deg$\downarrow$, $k_{\text{KNN}}\uparrow$}).
\hd{Wrong-way CV} anything using $y$ (feature screen, supervised PCA) \emph{must} be \emph{inside} each fold. Else CV biased $\downarrow$ (toward $0\%$ on noise w/ $p\!\gg\! n$).
\hd{Nested CV} inner = tuning, outer = honest assessment. Naive $\min_\theta\text{CV}_k$ is winner's-curse biased $\downarrow$.
Random $k$-fold \textbf{breaks} on time series $\Rightarrow$ rolling-origin / blocked. Validation set $\ne$ 2-fold.
\hd{Bootstrap} sample $n$ \emph{with replacement}; $\widehat{\text{SE}}=\text{sd}(\hat\theta^{*b})$. $B\!\in\![10^3,10^4]$.
$P(i\in\text{boot})=1-(1-1/n)^n\to 1-1/e\!\approx\!0.632$; OOB $\to 1/e\!\approx\!0.368$.
Percentile $95\%$ CI: $[\hat\theta^{*}_{(.025)},\hat\theta^{*}_{(.975)}]$ (order stats, \tr{not $\hat\theta\pm1.96\text{SE}$}). Quantifies \emph{variance}, \emph{not} bias. Paired (case) bootstrap for random-$X$/observational; residual bootstrap needs correct mean and homoscedastic resids. Boot-SE $\ne$ GLM-SE $\Rightarrow$ GLM assumptions violated, \emph{not} a bootstrap flaw.

\section{Ridge / Lasso / PCR}
Ridge $\min\|y-X\beta\|^2+\lambda\sum_{j\ge1}\beta_j^2$ (intercept \emph{unpenalised}); closed form $(X^\top X+\lambda I)^{-1}X^\top y$ on centered $X$.
Lasso $\min\tfrac{1}{2n}\|\cdots\|^2+\lambda\|\beta\|_1$; corners on axes $\Rightarrow$ exact zeros.
\hd{Geometry} $L^2$ ball smooth $\Rightarrow$ splits collinear pair; $L^1$ diamond $\Rightarrow$ corner picks one. \tr{Lasso path depends on dummy reference}; use group lasso for factors.
\hd{Std before:} ridge, lasso, PCA, PCR, KNN, $k$-means, hier.\ clust. OLS doesn't need it.
\hd{Subset counts} best subset $2^p$; forward (full path) $1+p(p+1)/2$; forward to size $K$: $1+\sum_{j=0}^{K-1}(p-j)$. Backward fails in $p>n$.
\hd{PCR $=$ discretised ridge} SVD: ridge shrinks PC-$j$ by $d_j^2/(d_j^2+\lambda)$ (smooth); PCR keeps top $M$ at weight 1, rest 0. PCR not variable selection. Lasso $>$ PCR for sparsity in original predictors.
\tr{Lasso $\ne$ AIC step}: avoids \emph{model-selection bias}; AIC+refit gives anti-conservative $p$-values; BIC does not fix it.
$p\!>\!n$ feasibility: ridge $\checkmark$, lasso $\checkmark$ (selects $\le n$), forward $\checkmark$, backward \tr{$\times$}, boosting $\checkmark$.

\section{Splines / GAM}
$\df$ counts (excl.\ global intercept): polynomial $d$: $d$; step $K$-cut: $K$; linear spline: $K+1$; \textbf{cubic spline: $K+3$}; \textbf{natural cubic: $K+1$}; smoothing spline: $\Tr(S_\lambda)\in[2,n]$, non-integer.
Cubic spline is $C^2$ at knots; \tr{3rd derivative discontinuous}; knots fixed in advance.
\hd{Smoothing spline penalty} $\int g''(t)^2dt$ -- on \emph{second} derivative, \tr{not first}.
$\lambda\uparrow\Rightarrow$ smoother, $\df\downarrow$; $\lambda\to\infty\Rightarrow\df=2$ (line). Opposite direction from poly degree / knot count.
\hd{GAM total $\df$} $1+\sum_j\df_j+\sum_c(L_c-1)$ (cat.\ levels).
\hd{GAM limitation} $\sum f_j(x_j)$ has \emph{no interactions} unless tensor-product added; trees of depth $d\ge2$ get them free. Boosting $\gg$ GAM gap $\Rightarrow$ interactions present.

\section{PCA}
Loadings $\phi_m$ unit-norm; score $z_{im}=\phi_m^\top x_i$ (no recentering if $x$ already standardised). Total var $=\sum\lambda_j=p$ on std.\ data; $\text{PVE}_m=\lambda_m/p$. PC1 \emph{maximises} variance; sign arbitrary.
\tr{NOT scale-invariant} -- largest-variance variable dominates PC1 if unscaled.
\tr{PCA blind to $y$} (PLS is the supervised cousin).

\section{Trees / Bagging / RF}
$\Var(\bar f_{\text{bag}})=\rho\sigma^2+\tfrac{1-\rho}{B}\sigma^2$ \tr{(prof-derived, not in ISLP)}; floor $\rho\sigma^2$ as $B\to\infty$.
\textbf{RF} decorrelates: $\mathtt{mtry}<p$ shrinks $\rho$. Defaults: $\sqrt p$ classif, $p/3$ regr.
$B$ \tr{NOT a tuning parameter} in RF: test err monotone $\downarrow$ in $B$; pick $\approx 500$-$1000$. Trees grown deep, no pruning. Bagging $=$ RF with $\mathtt{mtry}=p$.
\hd{OOB} each tree excludes $\approx 1/e\approx 37\%$; predict $i$ from only trees that didn't see $i$ $\Rightarrow$ near-unbiased test-err estimate, free.
\hd{Variable importance} magnitude only \tr{(no direction, shape, causality)}; pair with PDP/ICE. Prof prefers \textbf{permutation} (drop in OOB acc after shuffle) over Gini-impurity (biased toward high-cardinality).
\hd{Regression tree leaf prediction}: mean of training $y_i$ in region.

\section{Boosting (AdaBoost / GB / XGBoost)}
\hd{AdaBoost} weight $w_i^{(m+1)}\propto w_i^{(m)}\exp(\alpha_m\mathbf{1}[y_i\ne G_m])$, renormalise.
$\alpha_m=\log\!\bigl(\tfrac{1-\text{err}_m}{\text{err}_m}\bigr)$ (or $\tfrac12\log$; consistent w/ slide). err$=0.5\Rightarrow\alpha=0$; err$>0.5\Rightarrow\alpha<0$ \tr{(vote auto-flips)}.
\textbf{Derivation (prof-flagged)}: exp loss $L=e^{-yf}$, $y\in\{-1,+1\}$. Split sum into correct/wrong: $J(\alpha)=W_C e^{-\alpha}+W_W e^{\alpha}$; $dJ/d\alpha=0\Rightarrow e^{2\alpha}=W_C/W_W=(1-\text{err})/\text{err}$. AdaBoost $\equiv$ forward-stagewise additive on exp loss.
\hd{Gradient boosting} init $\hat f^{(0)}=\bar y$; fit tree to neg-grad $r_i=-\partial L/\partial f=y-f^{(m-1)}$ (squared loss $\Rightarrow$ residuals); update $\hat f^{(m)}=\hat f^{(m-1)}+\nu\hat T_m$.
\textbf{Hyperparams}: $M$ (\emph{IS} tuned, \tr{overfits} unlike RF); $d\in\{1\!-\!8\}$ (interaction order; $d=1$ = additive); $\nu\in[10^{-3},10^{-1}]$. \textbf{Coupling}: $M\cdot\nu\approx$ const; halving $\nu$ doubles required $M$.
\hd{Bagging vs boosting} bagging/RF reduce \emph{variance}; boosting reduces \emph{bias} (sequential). RF averages independent bootstraps; boosting sums correlated weak learners.
\hd{XGBoost} $\Omega(T)=\gamma|T|+\tfrac12\lambda\|w\|^2+\alpha\|w\|_1$; gradient \emph{and} Hessian (Newton/2nd-order); $\alpha$ drives small leaves to 0; $\gamma$ rejects splits unless gain $>\gamma$; row+col subsampling.

\section{Neural networks}
\hd{Param count} per layer $(n_{\text{in}}{+}1)\cdot n_{\text{out}}$ (\emph{bias on receiving} layer; inputs no bias). E.g.\ $4\!\to\!5\!\to\!3\!\to\!1 = 25{+}18{+}4=47$. Dropout / early stop / label smooth add 0 params.
Need non-linear activation (else $W_L\!\cdots\!W_1$ collapses to single affine $\Rightarrow$ linear regression).
\hd{ReLU} $\max(0,z)$; $g'(z)=\mathbf{1}[z>0]$ \tr{(dying ReLU)}. Sigmoid: $\sigma'=\sigma(1-\sigma)$.
\hd{Backprop seed} $\delta^{\text{out}}=-(y-\hat f)$ (squared loss). \textbf{Sigmoid + BCE} $\partial L/\partial z=\hat p-y$ (the $\hat p(1-\hat p)$ factors cancel $\Rightarrow$ no vanishing at saturation). Softmax + CE: $\delta^{\text{out}}_c=\hat p_c-y_c$. Chain for 1 hidden: $\partial L/\partial\alpha_{kj}=(\hat p-y)\beta_k g'(z_k)x_j$.
Backprop = \emph{gradient computation}; SGD/Adam does \emph{update}. Needs DAG (RNN $\to$ BPTT).
\hd{Mini-batch SGD} $\E[\widehat{\nabla L}]=\nabla L$ (unbiased); $\Var\propto 1/m$. Bigger batch $\Rightarrow$ less noise $\Rightarrow$ \emph{weaker} implicit reg. Pow-2 batch is hardware, not statistics. Implicit min-$\|\theta\|_2$ interpolator $\Rightarrow$ benign overfitting / flat minima (\tr{prof hobbyhorse}).
\hd{Learning rate} $\eta\in[10^{-3},10^{-1}]$ typical; $\eta=0.1$ fine; $\eta\approx 2$ diverges.
\hd{Dropout} train-only; rate $p\in[0.2,0.5]$, default 0.2; \tr{NEVER 50\% on tabular} (prof's iron rule); test = full net w/ rescaling.
\hd{Label smoothing} target $\tilde y_c=1-\varepsilon$ true, $\varepsilon/(C-1)$ else; $\varepsilon\!\approx\!0.05$. Motivation: label noise; discourages over-confident logits.
\hd{Early stopping} on \emph{validation} loss (\tr{not training}); return params at val-min epoch.
\hd{L2 weight decay} raises training err, lowers test err. ``Always regularise NN'' (SGD noise counts).

\section{Classifier metrics / ROC}
Sens $=$ TP$/$P $=$ TPR; Spec $=$ TN$/$N $=$ TNR; FPR $=1-$Spec; Err $=$ (FP$+$FN)$/n$; Prec $=$ TP$/$(TP$+$FP).
\hd{ROC} $y$-axis TPR, $x$-axis FPR $=1-$Spec; diag $=$ chance.
Threshold $\downarrow\Rightarrow$ more positives flagged $\Rightarrow$ Sens$\uparrow$, Spec$\downarrow$, point moves \textbf{NE}.
AUC $=P(\hat s(X^+)>\hat s(X^-))$; threshold-free. AUC $=0.5$ random; $<0.5$: flip predictions.
\hd{Imbalance} naive baseline acc $=\max\pi_k$ ($\Rightarrow$ accuracy lies). Use (Sens, Spec), AUC, PR-AUC, F1. Asymmetric cost (medical screening): optimise \emph{sensitivity}, lower threshold (\tr{0.5 inappropriate at base rate} 0.2/0.8).
\hd{KNN} std features (Euclid dominated by big-unit feat.); large $K\Rightarrow$ high bias.

\section{Clustering}
\hd{Hierarchical} complete $D(A,B)=\max d$; single $=\min$; average $=$ mean. \textbf{First merge is linkage-independent} (singletons). Recipe: find min off-diag $\to$ merge $\to$ recompute by rule. Heights non-decreasing. Cut at $h$: \#vertical lines crossed $=$ \#clusters. Leaf horizontal order arbitrary. \tr{$-1$P per labelled mistake} (incl.\ missing $y$-axis). Complete fuses $\ge$ single height for same pair.
\hd{$K$-means} non-increasing WCSS, local min (random init $\to$ many starts). NOT nested across $K$. Std features. User chooses $K$.

\section{Misc / direction-of-effect}
\tr{F-test}: joint test on $K-1$ dummies for a $K$-level factor (use \texttt{anova}; not individual $t$'s). Prof: \emph{name} it, don't compute mechanics. Individual $t$'s can all be insig.\ while joint $F$ rejects (collinearity).
\tr{Causal vs correlation}: prof slogan ``fancy correlations, not causal'' -- observational $\hat\beta$ are conditional associations, no matter how tiny the $p$.
\tr{Validation set $\ne$ 2-fold} (fits once vs twice).
\tr{Practical $\ne$ statistical} significance.
Reducible (var of $\hat f$, can be cut by $n$) vs irreducible $\sigma^2$ (cannot).
\hd{Plot diagnostics} (1) resid vs fit: zero-mean + homoscedasticity; (2) QQ: normality (\tr{prof favorite}); (3) scale-location: het.\ in std-resid form; (4) resid vs leverage: influence (high lev + high resid = danger).

\section{Worked numerics}
\hd{NN params} $4{\to}5{\to}3{\to}1$: $(5){\cdot}5+(6){\cdot}3+(4){\cdot}1=25{+}18{+}4=47$. \quad $128{\to}32{\to}64{\to}5$: $129{\cdot}32+33{\cdot}64+65{\cdot}5=6533$. Forgetting biases = canonical wrong.\\
\hd{AdaBoost $\alpha$} err$=0.20\Rightarrow\alpha=\log4\!\approx\!1.386$. err$=0.40\Rightarrow\alpha\!\approx\!0.405$. err$=0.70\Rightarrow\alpha\!\approx\!-0.847$ (flip). Weight mult: correct $\times e^{-\alpha}$, wrong $\times e^{+\alpha}$, renormalise.\\
\hd{OOB} $n{=}5$: $(4/5)^5{=}0.328$. $n{=}1000$: $\approx 1/e$, OOB count $\approx 368$.\\
\hd{Sigmoid} $\sigma(0.4)\!=\!0.599$; $\sigma(-1)\!=\!0.269$. Odds$=2.5\to p=5/7\!\approx\!0.714$. $p=0.3\to$ odds$=3/7$. $\exp(0.8)\!\approx\!2.225$.\\
\hd{Quadratic vertex} $\hat y=\beta_0+\beta_1 x+\beta_2 x^2$: minimum/max at $x^\star=-\beta_1/(2\beta_2)$ (min if $\beta_2{>}0$).\\
\hd{CI shortcut} large $\df$: $\hat\beta\pm 1.96\,\widehat{\text{SE}}$.

\section{Tuning / decision tables}
\hd{Standardise before?} \textbf{Yes}: ridge, lasso, PCA/PCR, KNN, $k$-means, hier.\ clust., SVM. \textbf{No}: OLS, plain logistic, trees/RF/boosting.\\
\hd{$p>n$ feasibility} ridge $\checkmark$, lasso $\checkmark$ ($\le n$ selected), forward step $\checkmark$, backward \tr{$\times$}, OLS \tr{$\times$}, GBM/RF/NN $\checkmark$.\\
\hd{Tuned by CV?} Ridge $\lambda$ $\checkmark$; Lasso $\lambda$ $\checkmark$; PCR $M$ $\checkmark$; PLS $M$ $\checkmark$; Smoothing-spline $\lambda$/df $\checkmark$; RF $\mathtt{mtry}$ $\checkmark$ (but $B$/ntree NO); GBM $M,\nu,d$ all $\checkmark$.\\
\hd{Method ranking gap} OLS$>$Ridge $\Rightarrow$ OLS variance-dominated; Ridge$<$GAM $\Rightarrow$ nonlinearities; GAM$<$Boost $\Rightarrow$ interactions present.\\
\hd{Sparsity} Lasso $\checkmark$; Ridge $\times$; PCR $\times$ (all original vars in every PC); Trees/Lasso/Best subset: yes.

\section{Plot reading / interpretation}
\hd{ROC curve} 45$^\circ$ diag = chance. Knee toward NW = better. AUC = area; two curves can have equal AUC and different shapes.
\hd{PCA biplot} arrows = loading vectors ($\phi_m$); long $y$-arrow $\Rightarrow$ high PC2 loading. Point on axis means: that observation's score on that PC. PC1 dominated by variable with largest $|\phi_{j1}|$.
\hd{Scree / cumPVE} elbow rule: keep PCs until cum.\ PVE plateaus, or to a target (80/90/95\%).
\hd{Coefficient table} sanity: $n-p-1=$ residual df. CI $\hat\beta\pm1.96\widehat{\text{SE}}$. Insig.\ $t$ + huge SE on a predictor with significant partner $\Rightarrow$ \emph{collinearity}, not ``no effect''.
\hd{Lasso path} coefs hit zero in order of weakness; reading sequence of zeros gives effective variable ranking.
\hd{CV plot} u-shape with min at $\lambda_{\min}$; band gives $\lambda_{1\text{SE}}$ (largest $\lambda$ within band $=$ simplest).

\section{Special cases / niche traps}
\hd{Mixture-class density} class-0 $=\sum_m w_m\phi(\cdot;\mu_m,\Sigma)$. $\log(f_0\pi_0)$ contains $\log\sum e^{\cdots}$ which \emph{doesn't} collapse $\Rightarrow$ Bayes boundary neither linear nor quadratic; LDA \emph{and} QDA both inadequate.
\hd{SVM ``linear in features''} only if every feature in the boundary appears in the listed basis. Cubic terms ($X_2^3$) break ``linear in $\{X_1,X_2,X_1^2,X_2^2\}$''. To classify a point: plug in, check sign.
\hd{KNN $K$} $K=1$: zero train err, max variance. $K=n$: constant prediction, max bias. Std features.
\hd{$K$-means} not nested across $K$ (\tr{going $K{\to}K{+}1$ can reshuffle globally}; only hier.\ is nested).
\hd{Sigmoid+SE vs Sigmoid+BCE} BCE cancels $p(1-p)$ in gradient $\Rightarrow$ no vanishing at saturation. Use BCE.
\hd{Boosting init} squared loss: $\hat f^{(0)}=\bar y$. Classification AdaBoost: weights $w_i^{(0)}=1/n$.
\hd{Linkage height ineq.} for same pair: $h_{\text{complete}}\ge h_{\text{average}}\ge h_{\text{single}}$.

\section{Per-question opening moves}
\hd{Read OLS table} (1) $n=$ resid df $+$ \#params; (2) CI $=\hat\beta\pm1.96\widehat{\text{SE}}$; (3) check interaction $\Rightarrow$ main effect is conditional; (4) joint test on factors via $F$; (5) collinearity if huge SE + sig.\ $F$.
\hd{Read logistic table} odds factor $=\exp(\beta\Delta)$; per-group multipliers $=\exp(\beta_x+\beta_{xz}z)$; $\hat\eta=\beta^\top x$ then $\sigma$; threshold vs base rate; class imbalance flags.
