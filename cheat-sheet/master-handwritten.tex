% Handwritten / ballpoint-pen variant.
% Compile with: lualatex master-handwritten.tex
% (Was xelatex; switched to lualatex to enable per-glyph / per-line jitter.)
% Inputs the same parts/ files as master.tex — do not duplicate content.

\documentclass[10pt,a4paper,twocolumn]{extarticle}

% ===== Geometry =====
\usepackage[a4paper,margin=12mm,columnsep=8mm]{geometry}

% ===== Math + tables =====
\usepackage{amsmath,amssymb,amsthm,mathtools}
\usepackage{bm}
\usepackage{booktabs}
\usepackage{array}
\usepackage{tabularx}
\usepackage{multirow}
\usepackage{makecell}

% ===== Handwriting font via fontspec (XeLaTeX) =====
\usepackage{fontspec}
\setmainfont{PatrickHand-Regular.ttf}[
  Path=./fonts/,
  Scale=1.10,
  Ligatures=TeX,
  BoldFont=PatrickHand-Regular.ttf,   % no separate bold; reuse regular
  ItalicFont=PatrickHand-Regular.ttf,
]
\setsansfont{PatrickHand-Regular.ttf}[Path=./fonts/]
% Monospace stays default — we use \texttt only for path refs.

% ===== Make math letters inherit the handwriting font =====
\usepackage[italic,defaultmathsizes]{mathastext}
% \Mathastext activates after preamble; Latin letters + digits in math now use Bradley Hand.
% Greek letters and operator symbols keep Computer Modern (still readable, looks math-y).

% ===== Ballpoint-blue ink color =====
\usepackage{xcolor}
\definecolor{ballpoint}{HTML}{1B2C7A}   % deep navy blue, classic biro
\AtBeginDocument{\color{ballpoint}}
% Force the algorithm / algorithmic envs to inherit ballpoint too
% (they open their own group and reset to black otherwise).
\usepackage{etoolbox}
\AtBeginEnvironment{algorithm}{\color{ballpoint}}
\AtBeginEnvironment{algorithmic}{\color{ballpoint}}
\makeatletter
\patchcmd{\@algocf@start}{\color{ballpoint}}{\color{ballpoint}}{}{} % no-op safety
\makeatother

% ===== Misc =====
\usepackage{microtype}
\usepackage[parfill]{parskip}
\usepackage{enumitem}
\setlist{nosep,leftmargin=1.2em}

\usepackage{algorithm}
\usepackage{algpseudocode}
\usepackage{listings}

% ===== Lighter, less "printed" section headings =====
\usepackage{titlesec}
\titleformat{\section}{\normalsize}{\thesection.}{0.4em}{}
\titlespacing{\section}{0pt}{0.6ex}{0.3ex}
\titleformat{\subsection}{\small}{\thesubsection}{0.4em}{}
\titlespacing{\subsection}{0pt}{0.4ex}{0.2ex}

% Soften booktabs rules — thin biro strokes, not engraver lines.
\setlength{\heavyrulewidth}{0.4pt}
\setlength{\lightrulewidth}{0.3pt}
\setlength{\cmidrulewidth}{0.3pt}
\setlength{\aboverulesep}{1pt}
\setlength{\belowrulesep}{1pt}


% ===== Convenience macros (must match master.tex) =====
\newcommand{\R}{\mathbb{R}}
\let\E\relax  % mathastext may clobber; redefine.
\DeclareMathOperator{\E}{\mathbb{E}}
\DeclareMathOperator{\Var}{Var}
\DeclareMathOperator{\Cov}{Cov}
\DeclareMathOperator{\Corr}{Corr}
\DeclareMathOperator{\Bias}{Bias}
\newcommand{\T}{^{\!\top}}
\newcommand{\xb}{\bm{x}}
\newcommand{\yb}{\bm{y}}
\newcommand{\bbeta}{\bm{\beta}}
\newcommand{\Xb}{\bm{X}}

\pagestyle{empty}
\setlength{\columnseprule}{0pt}
\raggedbottom

% Load the per-glyph + per-line jitter callbacks.
\directlua{dofile("jitter.lua")}

\begin{document}
